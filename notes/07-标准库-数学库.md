# 数学库（math.h）

- VS 中调用 `M_PI` 需要先定义 `_USE_MATH_DEFINES`。

```c
#define _USE_MATH_DEFINES
#include <stdio.h>
#include <math.h>

int main(void) {

	double num = 1.0;
	sin(num); // 正弦函数
	asin(num); // 反正弦函数
	sinh(num);	// 双曲正弦函数
	exp(num); // 以 e 为底的指数函数
	log(num); // 以 e 为底的对数函数
	log2(num); // 以 2 为底的对数函数
	log10(num); // 以 10 为底的对数函数
	sqrt(num); // 平方根函数
	fabs(num); // 绝对值函数
	ceil(num); // 向上取整函数
	floor(num); // 向下取整函数
	fmod(num, 0.5); // 取模函数
	pow(num, 2.0); // 幂函数

	printf("The value of M_PI is: %f\n", M_PI);

	return 0;
}
```
